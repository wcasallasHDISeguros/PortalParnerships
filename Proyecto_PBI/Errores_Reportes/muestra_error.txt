let
    Origen = AmazonRedshift.Database(
        "corshftanltc-dprogramp.hdicolombia.com.co:9519",
        "liberty",
        [ProviderName=null, BatchSize=null]
    ),

    Consulta = Value.NativeQuery(
        Origen,
        "
        WITH cliente_unico AS (
            SELECT
                razon_social,
                nombre_1,
                nombre_2,
                apellido_1,
                MIN(apellido_2) AS apellido_2,
                nro_unico_persona
            FROM liberty.cliente.dwh_cliente_unico
            GROUP BY
                razon_social,
                nombre_1,
                nombre_2,
                apellido_1,
                nro_unico_persona
        ),

        tabla1 AS (
            SELECT
                a.sseguro AS ""SSEGURO"",
                a.ramo_prod AS ""RAMO_PROD"",
                a.periodo_contable AS ""PERIODO_CONTABLE"",
                a.sbu AS ""SBU"",
                a.usuario_mov AS ""USUARIO_MOV"",
                a.intermediario_lide AS ""INTERMEDIARIO_LIDE"",
                a.sucursal_prod AS ""SUCURSAL_PROD"",
                a.nro_identifi_tom AS ""NRO_IDENTIFI_TOM"",
                a.fi_certificado AS ""FI_CERTIFICADO"",
                a.fi_anexo AS ""FI_ANEXO"",
                a.poliza AS ""POLIZA"",
                a.certificado AS ""certificado"",
                a.recibo AS ""RECIBO"",

                SUM(a.vr_prima_mn_orig) AS ""VR_PRIMA_MN_ORIG"",
                SUM(a.vr_prima_pesos_coa) AS ""VR_PRIMA_PESOS_COA"",

                a.tipo_transac AS ""TIPO_TRANSAC"",
                a.cod_modalidad AS ""COD_MODALIDAD"",
                a.frec_pago AS ""frec_pago"",

                CASE
                    WHEN a.sucursal_prod = 177
                         AND a.ramo_prod = '900753'
                        THEN 'COOMEVA'

                    WHEN a.sucursal_prod = 226
                        THEN 'CREDIORBE'

                    WHEN a.sucursal_prod = 214
                         AND a.ramo_prod = '900753'
                        THEN 'BANCO DE OCCIDENTE'

                    WHEN a.sucursal_prod IN (200,230,231)
                         AND a.ramo_prod IN (
                             '900753','6041','6042',
                             '6045','6046','6047','6049'
                         )
                        THEN 'OTROS BANCOS'

                    WHEN a.ramo_prod IN ('6039')
                         AND a.sucursal_prod NOT IN (
                             177,200,214,230,231,226
                         )
                        THEN 'PESADOS INDIVIDUAL'

                    WHEN a.ramo_prod IN ('6038','6046')
                         AND a.sucursal_prod NOT IN (
                             177,200,214,230,231,226
                         )
                        THEN 'TAXIS'

                    WHEN a.ramo_prod IN ('6047','6049')
                         AND a.sucursal_prod NOT IN (
                             177,200,214,230,231,226
                         )
                        THEN 'PESADOS COLECTIVO'

                    WHEN a.ramo_prod IN (
                             '6041','6042','6045','900753'
                         )
                         AND a.sucursal_prod NOT IN (
                             177,200,214,230,231,226
                         )
                        THEN 'LIVIANOS'

                    ELSE 'PRIVATE'
                END AS ""SEGMENTO"",

                CASE
                    WHEN t3.razon_social = '' THEN
                        COALESCE(TRIM(t3.nombre_1), '') || ' ' ||
                        COALESCE(TRIM(t3.nombre_2), '') || ' ' ||
                        COALESCE(TRIM(t3.apellido_1), '') || ' ' ||
                        COALESCE(TRIM(t3.apellido_2), '')
                    ELSE t3.razon_social
                END AS ""NOMBRE_TOM""

            FROM liberty.prod.dwh_polizas_h a

            LEFT JOIN cliente_unico t3
                ON a.nro_unico_tomador = t3.nro_unico_persona

            WHERE
                a.periodo_contable >= 202301
                AND a.periodo_contable <= 202412
                AND a.sbu = 'AUT'

            GROUP BY
                a.sseguro,
                a.ramo_prod,
                a.periodo_contable,
                a.sbu,
                a.usuario_mov,
                a.intermediario_lide,
                a.sucursal_prod,
                a.nro_identifi_tom,
                a.fi_certificado,
                a.fi_anexo,
                a.poliza,
                a.certificado,
                a.recibo,
                a.tipo_transac,
                a.cod_modalidad,
                a.frec_pago,
                t3.razon_social,
                t3.nombre_1,
                t3.nombre_2,
                t3.apellido_1,
                t3.apellido_2
        ),

        tabla2 AS (
            SELECT
                sseguro,
                trespue
            FROM (
                SELECT
                    a.sseguro,
                    a.cpregun,
                    a.crespue,
                    a.nmovimi,
                    b.trespue,
                    ROW_NUMBER() OVER (
                        PARTITION BY a.sseguro
                        ORDER BY a.nmovimi DESC
                    ) AS indice

                FROM liberty.axis.pregunpolseg a

                LEFT JOIN liberty.axis.respuestas b
                    ON a.cpregun = b.cpregun
                    AND a.crespue = b.crespue
                    AND b.cidioma = 8

                WHERE a.cpregun = 4790
            ) t0
            WHERE indice = 1
        ),

        tabla3 AS (
            SELECT
                sseguro,
                cmatric
            FROM (
                SELECT
                    sseguro,
                    nmovimi,
                    cmatric,
                    ROW_NUMBER() OVER (
                        PARTITION BY sseguro
                        ORDER BY nmovimi DESC
                    ) AS indice
                FROM liberty.axis.autriesgos
            ) t1
            WHERE indice = 1
        )

        SELECT
            tabla1.*,
            tabla2.trespue AS ""TRESPUE"",
            tabla3.cmatric AS ""CMATRIC"",
            tabla5.oficina_fusion AS ""OFICINA_FUSION""

        FROM tabla1

        LEFT JOIN tabla2
            ON tabla1.""SSEGURO"" = tabla2.sseguro

        LEFT JOIN tabla3
            ON tabla1.""SSEGURO"" = tabla3.sseguro

        LEFT JOIN liberty.as400.maestro_intermediarios tabla4
            ON tabla1.""INTERMEDIARIO_LIDE"" = tabla4.interm

        LEFT JOIN liberty.apoyo.dwh_redcomercial_sucursal tabla5
            ON tabla4.sucursal = tabla5.sucursal
        ",
        null,
        [EnableFolding=false]
    )
in
    Consulta
