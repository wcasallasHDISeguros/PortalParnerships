        "
        SELECT
            TRIM(CAST(w8racg AS VARCHAR)) || '-' ||
            CAST(w8ansn AS VARCHAR) || '-' ||
            CAST(w8scpr AS VARCHAR) || '-' ||
            CAST(w8rcnu AS VARCHAR) AS ""SINIESTRO"",

            '1' ||
            LEFT(CAST(w8racg AS VARCHAR) || ' ', 3) ||
            CAST(w8ansn AS VARCHAR) ||
            RIGHT('000' || CAST(w8scpr AS VARCHAR), 3) ||
            RIGHT('0000000' || CAST(w8rcnu AS VARCHAR), 7)

            w8racg AS ""RAMO_PROD"",
            w8ansn AS ""ANNO_RADICACION"",
            w8scpr AS ""SUCURSAL_PROD"",
            w8rcnu AS ""RADICACION"",

            w8tdnb AS ""NOMBRE_TOM"",
            w8asnt AS ""NIT_ASEG"",
            w8asnb AS ""NOMBRE_ASEG"",
            w8vhph AS ""PLACA"",
            w8vhmk AS ""Marca_Vehiculo"",
            w8vhml AS ""Modelo_Vehiculo"",
            w8mrnu AS ""Numero_Motor""

        WHERE w8racg IN (
            '6031', '6032', '6033', '6034', '6035',
            '6036', '6037', '6038', '6039', '6040',
            '6041', '6042', '6043', '6044', '6045',

            'MT', 'MU', 'MW', 'MWT', 'PE',
            'PI', 'PM', 'PO', 'TI', 'TP',
            'TT', 'TU', 'TO', '900753', '900792'
        )
        ",
        null,
        [EnableFolding=false]
    )

    Consulta
