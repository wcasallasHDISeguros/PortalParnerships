let
    Origen = AmazonRedshift.Database(
        "corshftanltc-dprogramp.hdicolombia.com.co:9519",
        "liberty",
        [ProviderName=null, BatchSize=null]
    ),

    Consulta = Value.NativeQuery(
        Origen,
        "
        SELECT
            llave_pol,
            nro_placa,
            RIGHT('00' || CAST(cod_marca AS VARCHAR), 3) ||
            RIGHT('00' || CAST(cod_clase AS VARCHAR), 2) ||
            RIGHT('00' || CAST(cod_tipo AS VARCHAR), 3)
                AS ""COD_Fasecolda"",
            CAST(
                CAST(ann_contable AS VARCHAR) ||
                RIGHT('00' || CAST(mes_contable AS VARCHAR), 2)
                AS INTEGER
            ) AS ""PER""
        FROM dt.dwh_dt_aut_autos
        WHERE ann_contable >= 1800
          AND CAST(
                CAST(ann_contable AS VARCHAR) ||
                RIGHT('00' || CAST(mes_contable AS VARCHAR), 2)
                AS INTEGER
              ) >= CAST(
                TO_CHAR(
                    DATEADD(month, -1000, GETDATE()),
                    'YYYYMM'
                ) AS INTEGER
              )
        ",
        null,
        [EnableFolding=false]
    )

in
    Consulta
