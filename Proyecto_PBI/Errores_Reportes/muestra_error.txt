let
    Origen = AmazonRedshift.Database(
        "corshftanltc-dprogramp.hdicolombia.com.co:9519",
        "liberty",
        [ProviderName=null, BatchSize=null]
    ),

    Consulta = Value.NativeQuery(
        Origen,
        "
        SELECT
            a.ramo_prod AS RAMO_PROD,
            a.periodo_contable AS PERIODO_CONTABLE,
            a.nro_identifi_tom AS NRO_IDENTIFI_TOM,
            a.sucursal_prod,
            a.poliza AS POLIZA,
            SUM(a.vr_prima_mn_orig) AS VR_PRIMA_MN_ORIG,
            SUM(a.vr_prima_pesos_coa) AS VR_PRIMA_PESOS_COA,
            SUM(a.vr_prima_anual) AS VR_PRIMA_ANUAL,
            CASE
                WHEN t3.razon_social = ''
                THEN
                    TRIM(COALESCE(t3.nombre_1, '')) || ' ' ||
                    TRIM(COALESCE(t3.nombre_2, '')) || ' ' ||
                    TRIM(COALESCE(t3.apellido_1, '')) || ' ' ||
                    TRIM(COALESCE(t3.apellido_2, ''))
                ELSE t3.razon_social
            END AS NOMBRE_TOM
        FROM prod.dwh_polizas_h a
        LEFT JOIN (
            SELECT
                razon_social,
                nombre_1,
                nombre_2,
                apellido_1,
                MIN(apellido_2) AS apellido_2,
                nro_unico_persona
            FROM cliente.dwh_cliente_unico
            GROUP BY
                razon_social,
                nombre_1,
                nombre_2,
                apellido_1,
                nro_unico_persona
        ) t3
            ON a.nro_unico_tomador = t3.nro_unico_persona
        WHERE
            a.periodo_contable BETWEEN 201909 AND 222222
            AND a.sbu = 'AUT'
            AND a.poliza IN (450195, 466192, 483535)
        GROUP BY
            a.ramo_prod,
            a.periodo_contable,
            a.nro_identifi_tom,
            a.sucursal_prod,
            a.poliza,
            t3.razon_social,
            t3.nombre_1,
            t3.nombre_2,
            t3.apellido_1,
            t3.apellido_2
        ",
        null,
        [EnableFolding=true]
    ),

    Resultado = Table.AddColumn(
        Table.TransformColumnTypes(
            Consulta,
            {
                {"PERIODO_CONTABLE", type text},
                {"POLIZA", type text}
            }
        ),
        "PC_poliza",
        each [PERIODO_CONTABLE] & "-" & [POLIZA],
        type text
    )
in
    Resultado
